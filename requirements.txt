# types.GoogleMaps, types.UrlContext и т.п.) активно используется в bot.py
# и менялся между мажорными версиями. Обновлять версии здесь — осознанное
# решение с последующей проверкой /diag и ручным тестом основных команд,
# а не побочный эффект случайного редеплоя.

# tzdata — python:3.10-slim (см. Dockerfile) не ставит системную базу часовых
# поясов через apt, а stdlib-модуль zoneinfo (используется в _current_quota_day
# для дневного сброса счётчиков квоты по America/Los_Angeles — там у Google
# полночь для обнуления RPD-лимитов) без нeё падает ZoneInfoNotFoundError. Код
# и так безопасно откатывается на UTC при отсутствии данных (см. try/except в
# _current_quota_day), но тогда сброс квоты сдвигается на 7-8 часов от реального
# обнуления лимита у Google — с этим пакетом сброс происходит в верное время.
# Чистый pip-пакет с данными IANA tzdb, без компиляции — не раздувает образ.
tzdata==2025.2
# sentry-sdk — опциональный (см. SENTRY_DSN в bot.py) персистентный трекинг ошибок:
# bot.log живёт на эфемерном диске HF Spaces и теряется при каждом редеплое (см.
# README, "Известные ограничения" — раньше это было единственным незакрытым
# пунктом там). Без SENTRY_DSN sentry_sdk.init() не вызывается вообще — пакет
# просто лежит неиспользуемым, поведение не меняется для тех, кто его не настроил.
sentry-sdk==2.66.1
